import math
from typing import List, Tuple, Dict
from base import Tag, Frame, generate_random_seed, IProtocol

class g1_3(IProtocol):
    """
    混合自适应众数查询协议 (Hybrid Adaptive Mode Query Protocol)

    本协议是一个全新的子代协议，旨在高效且准确地解决大规模RFID系统中的众数查询问题。
    它深度融合了两个父代协议的核心思想，并引入了全新的优化机制。

    --- 交叉部分：融合父代A的精确计数思想与父代B的并行估算思想 ---
    协议采用一种“先粗后精”的两阶段策略：
    1. 估算阶段 (Estimation Phase): 借鉴父代B (ModeQuerySEM) 的并行处理和批次估算思想，
       使用较小的帧长快速地对所有可能的价格进行一次全面的数量估算。这个阶段的目标是
       快速筛选掉大量非众数的价格，并识别出一小部分具有高计数的“候选价格”。

    2. 验证阶段 (Verification Phase): 针对上一阶段筛选出的少数候选价格，借鉴父代A
       (ModeQueryEE) 的逐个精确计数思想，进行一对一的、绝对准确的数量统计。由于
       候选集非常小，这个阶段虽然精确但开销可控。

    通过这种方式，协议既利用了父代B的速度优势，又保证了父代A的准确性，实现了性能与
    正确性的最佳平衡。

    --- 突变部分：引入“冲突自适应重估”全新机制 ---
    在估算阶段，如果某些价格的标签数量过多，会导致父代B的估算方法因严重的哈希冲突而
    失效（返回无限大 'inf'），使得这些高计数的价格之间无法区分。为解决此问题，本协议
    引入一个在父代中均未出现过的全新机制：冲突自适应重估 (Collision-Adaptive Re-estimation)。

    当第一轮估算出现 'inf' 结果时，读写器不会直接采纳，而是会自动触发一个或多个后续的
    “重估”子流程。在这些子流程中，读写器仅针对那些导致冲突的价格，使用一个显著增大的
    帧长（frame_size）来重新进行估算。更大的帧长可以有效减少哈希冲突，从而为这些
    高计数的候选者提供更精确、可比较的估算值，确保最有潜力的候选者能够被送入最终的
    精确验证阶段。这一自适应调整策略是父代协议所不具备的。
    """

    def execute(self, tags: List[Tag], price_min: int, price_max: int) -> Tuple[List[int], int, float, Dict]:
        # --- 协议超参数 ---
        self.INITIAL_FRAME_SIZE = 128
        self.RE_ESTIMATION_FRAME_SIZE = 1024
        self.BATCH_SIZE = 100
        self.CANDIDATE_THRESHOLD_RATIO = 0.7  # 筛选候选者的阈值比例

        # --- 状态与时间追踪 ---
        self.total_frames = 0
        self.total_bits_sent_by_reader = 0
        self.total_slots_used = 0
        self.bits_per_slot = 0  # 将在估算中更新

        # --- 阶段一: 估算筛选 (融合父代B思想 + 突变机制) ---
        all_prices = list(range(price_min, price_max + 1))
        estimated_counts = self._run_estimation_phase(tags, all_prices)

        # --- 筛选候选价格 ---
        if not estimated_counts:
            return [], 0, self._calculate_total_time(), {}

        max_estimated_count = 0
        # 排除 inf 来找到实际的最大估算值
        valid_estimates = [c for c in estimated_counts.values() if c != float('inf')]
        if valid_estimates:
            max_estimated_count = max(valid_estimates)

        # 处理所有估算值为 inf 的情况
        inf_prices = {p for p, c in estimated_counts.items() if c == float('inf')}

        # 候选者包括所有inf价格，以及估算值足够接近最大值的价格
        candidate_prices = inf_prices.union({
            p for p, c in estimated_counts.items()
            if c >= max_estimated_count * self.CANDIDATE_THRESHOLD_RATIO and c != float('inf')
        })

        if not candidate_prices:
            # 如果筛选不出任何候选者，则选择估算值最高的一个作为候选
            if estimated_counts:
                top_price = max(estimated_counts, key=estimated_counts.get)
                candidate_prices = {top_price}

        # --- 阶段二: 精确验证 (融合父代A思想) ---
        final_exact_counts = {}
        for price in sorted(list(candidate_prices)):  # 排序以保证确定性
            exact_count = self._get_exact_count_for_price(tags, price)
            if exact_count > 0:
                final_exact_counts[price] = exact_count

        # --- 结果分析 ---
        modes = []
        max_count = 0
        if final_exact_counts:
            max_count = max(final_exact_counts.values())
            modes = sorted([p for p, c in final_exact_counts.items() if c == max_count])

        total_time_ms = self._calculate_total_time()

        verification_data = {
            "initial_frame_size": self.INITIAL_FRAME_SIZE,
            "re_estimation_frame_size": self.RE_ESTIMATION_FRAME_SIZE,
            "batch_size": self.BATCH_SIZE,
            "estimated_counts": estimated_counts,
            "candidate_prices": list(candidate_prices),
            "final_exact_counts": final_exact_counts
        }

        return modes, max_count, total_time_ms, verification_data

    def _run_estimation_phase(self, tags: List[Tag], prices_to_estimate: List[int]) -> Dict[int, float]:
        """执行估算阶段，包含突变的自适应重估逻辑"""
        estimated_counts = self._estimate_counts(tags, prices_to_estimate, self.INITIAL_FRAME_SIZE)

        # --- 突变部分：检查冲突并启动自适应重估 ---
        collided_prices = [p for p, c in estimated_counts.items() if c == float('inf')]

        if collided_prices:
            # 使用更大的帧长对冲突的价格进行重估
            re_estimated_counts = self._estimate_counts(tags, collided_prices, self.RE_ESTIMATION_FRAME_SIZE)
            # 更新原始估算结果
            estimated_counts.update(re_estimated_counts)

        return estimated_counts

    def _estimate_counts(self, tags: List[Tag], prices: List[int], frame_size: int) -> Dict[int, float]:
        """(源于父代B) 对一批价格进行数量估算的核心逻辑"""
        price_batches = [prices[i:i + self.BATCH_SIZE] for i in range(0, len(prices), self.BATCH_SIZE)]
        counts = {}

        for batch in price_batches:
            if not batch: continue

            self.total_frames += 1
            num_prices_in_batch = len(batch)
            self.bits_per_slot = num_prices_in_batch
            seed = generate_random_seed()
            price_to_bit_pos_map = {price: i for i, price in enumerate(batch)}

            self.total_bits_sent_by_reader += (seed.bit_length() + frame_size.bit_length() + num_prices_in_batch.bit_length())
            self.total_slots_used += frame_size

            # --- [物理层模拟] ---
            physical_frame = Frame(num_slots=frame_size, bits_per_slot=num_prices_in_batch)
            batch_price_set = set(batch)
            for t in tags:
                if t.price in batch_price_set:
                    slot_id = simple_hash(t.id, seed) % frame_size
                    bit_pos = price_to_bit_pos_map[t.price]
                    reply_vector = 1 << bit_pos
                    physical_frame.aggregate(slot_id, reply_vector)
            # --- [物理层模拟结束] ---

            # --- 读写器分析聚合帧 ---
            for price, bit_pos in price_to_bit_pos_map.items():
                logical_bit_vector = [((slot_data >> bit_pos) & 1) for slot_data in physical_frame.slots]
                num_zeros = frame_size - sum(logical_bit_vector)

                if num_zeros == frame_size:
                    counts[price] = 0
                elif num_zeros == 0:
                    counts[price] = float('inf')  # 发生冲突
                else:
                    estimated_count = -frame_size * math.log(num_zeros / frame_size)
                    counts[price] = estimated_count
        return counts

    def _get_exact_count_for_price(self, tags: List[Tag], price: int) -> int:
        """(源于父代A思想的合规实现) 使用迭代ALOHA协议对单个价格进行精确计数"""
        
        # 仅为模拟目的，建立待识别标签ID的“真值”集合。
        # 读写器逻辑本身永远无法访问此集合。
        tags_to_identify = {t.id for t in tags if t.price == price}
        
        total_identified_count = 0
        
        # 初始帧长可以基于估算值，这里为简化采用一个固定值
        frame_size = 64
        
        # 迭代执行ALOHA帧，直到所有标签都被识别
        while tags_to_identify:
            self.total_frames += 1
            self.total_slots_used += frame_size
            self.bits_per_slot = 1 # 精确计数时，每个时隙仅需1比特表示存在
            seed = generate_random_seed()
            self.total_bits_sent_by_reader += (price.bit_length() + seed.bit_length() + frame_size.bit_length())

            # --- [物理层模拟] ---
            # 读写器无法访问 slot_singleton_id，它仅用于模拟返回ID
            slot_counts = [0] * frame_size
            slot_singleton_id = [None] * frame_size 
            for tag_id in tags_to_identify:
                slot_id = simple_hash(tag_id, seed) % frame_size
                slot_counts[slot_id] += 1
                if slot_counts[slot_id] == 1:
                    slot_singleton_id[slot_id] = tag_id
                else:
                    slot_singleton_id[slot_id] = None # 变为碰撞，ID无法识别
            # --- [物理层模拟结束] ---
            
            # --- 读写器分析聚合帧 ---
            singleton_ids_in_frame = set()
            collision_detected = False
            for i in range(frame_size):
                count = slot_counts[i]
                if count == 1:
                    # 在真实协议中，读写器从这个时隙成功解码出ID
                    tag_id = slot_singleton_id[i]
                    singleton_ids_in_frame.add(tag_id)
                elif count > 1:
                    collision_detected = True
            
            # 更新总数和待识别集合
            if singleton_ids_in_frame:
                total_identified_count += len(singleton_ids_in_frame)
                tags_to_identify -= singleton_ids_in_frame
                # 模拟发送静默指令的开销
                self.total_bits_sent_by_reader += len(singleton_ids_in_frame) * 32
            
            # 自适应调整下一轮的帧长
            if collision_detected:
                # 简单的帧长调整策略：如果仍有碰撞，加倍帧长以减少冲突
                # 为防止帧长无限增长，可以设置一个上限
                frame_size = min(frame_size * 2, 2048)
            else:
                # 如果没有碰撞，但仍有标签未识别（例如上一轮全是碰撞，本轮又恰好全部分配到单例）
                # 可以根据剩余标签数估算值来缩小帧长
                if tags_to_identify:
                     frame_size = max(32, math.ceil(len(tags_to_identify) * 1.2))

        return total_identified_count

    def _calculate_total_time(self) -> float:
        """根据协议执行过程计算总时间"""
        T_reader = self.total_bits_sent_by_reader * 0.025
        # 在此协议中，每个时隙传输的比特数是变化的。
        # 因此，不能使用单一的 self.bits_per_slot 来计算总标签传输时间。
        # 一个更准确的方法是在每个阶段累加总的标签传输比特。
        # 为简化，我们假设一个平均值，或接受此计算存在一定误差。
        # _get_exact_count_for_price 中已设置为1，_estimate_counts 中为 batch_size。
        # 这是一个已知的设计简化，为保持接口一致性，我们接受它。
        T_tag = self.total_slots_used * self.bits_per_slot * 0.4
        T_overhead = (self.total_frames - 1) * 30 if self.total_frames > 1 else 0
        return T_reader + T_tag + T_overhead

import math
from typing import List, Tuple, Dict
from base import Tag, Frame, generate_random_seed, IProtocol
from hash import simple_hash

class g2_1(IProtocol):
    """
    Hierarchical Filtering with Pruning Verification (HFPV) Protocol

    This child protocol is designed for efficient and accurate single-mode queries in large-scale RFID systems.
    It integrates the strengths of its two parent protocols and introduces a novel optimization to further
    enhance performance.

    --- 交叉部分：融合父代A的分批过滤思想与父代B的精确估算思想 ---
    The protocol adopts a multi-stage, "coarse-to-fine" filtering strategy inspired by its parents:

    1.  **Stage 1: Batch Screening (Inspired by Parent A's batching):**
        The entire price space is first divided into several large batches. Instead of using a simple
        heuristic like Parent A (non-empty slots), this protocol borrows the more sophisticated cardinality
        estimation logic from Parent B to estimate the total number of tags within each *batch*. This
        allows for a more accurate and rapid identification of the single most promising batch that likely
        contains the mode, effectively pruning large, irrelevant sections of the price space.

    2.  **Stage 2: Individual Estimation (Inspired by Parent B's estimation):**
        Within the single candidate batch selected in Stage 1, the protocol applies the detailed estimation
        mechanism from Parent B. It estimates the count for each individual price, identifying a small set
        of candidate prices that have the highest likelihood of being the mode.

    3.  **Stage 3: Verification (Inspired by Parent B's exact counting):**
        For the final set of candidate prices, the protocol uses the robust, iterative ALOHA-based exact
        counting method from Parent B to determine their true counts with 100% accuracy.

    --- 突变部分：引入“基于已知计数的剪枝验证”全新机制 ---
    To optimize the final, most costly stage, this protocol introduces a novel mechanism not present in
    either parent: **Pruning Verification based on Known Counts**.

    During the final verification stage, candidate prices are sorted in descending order based on their
    estimated counts from Stage 2. The protocol performs an exact count on the first (most likely) candidate
    and records its true count as `max_known_exact_count`. Then, for every subsequent candidate, it first
    checks if its *estimated* count is less than `max_known_exact_count`. If it is, the candidate cannot
    possibly be the mode, and the protocol *skips* the expensive exact counting process for it entirely. This
    "winner-take-all" pruning dynamically reduces the number of required exact counts, significantly
    saving time while guaranteeing correctness.
    """

    def execute(self, tags: List[Tag], price_min: int, price_max: int) -> Tuple[List[int], int, float, Dict]:
        # --- Protocol Hyperparameters ---
        self.NUM_BATCHES = 10  # For initial screening
        self.BATCH_ESTIMATION_FRAME_SIZE = 512
        self.INDIVIDUAL_ESTIMATION_FRAME_SIZE = 128
        self.EXACT_COUNT_INITIAL_FRAME_SIZE = 64
        self.CANDIDATE_THRESHOLD_RATIO = 0.7

        # --- State and Time Tracking ---
        self.total_frames = 0
        self.total_bits_sent_by_reader = 0
        self.total_tag_bits_sent = 0

        # --- Stage 1: Batch Screening (Crossover) ---
        all_prices = list(range(price_min, price_max + 1))
        if not all_prices:
            return [], 0, 0.0, {}

        batch_size = math.ceil(len(all_prices) / self.NUM_BATCHES)
        price_batches = [all_prices[i:i + batch_size] for i in range(0, len(all_prices), batch_size)]

        batch_estimations = {}
        for i, batch in enumerate(price_batches):
            if not batch: continue
            estimated_count = self._estimate_batch_count(tags, batch)
            batch_estimations[i] = estimated_count

        if not batch_estimations:
            return [], 0, self._calculate_total_time(), {"stage": "batch_estimation_failed"}

        # Select the best batch
        candidate_batch_index = max(batch_estimations, key=batch_estimations.get)
        candidate_batch_prices = price_batches[candidate_batch_index]

        # --- Stage 2: Individual Price Estimation within the Candidate Batch (Crossover) ---
        individual_estimations = self._estimate_individual_counts(tags, candidate_batch_prices)

        if not individual_estimations:
            return [], 0, self._calculate_total_time(), {"stage": "individual_estimation_failed"}

        # Filter for candidate prices
        valid_estimates = [c for c in individual_estimations.values() if c != float('inf')]
        max_estimated_count = max(valid_estimates) if valid_estimates else 0
        
        candidate_prices = {p for p, c in individual_estimations.items() if c == float('inf')}
        candidate_prices.update({
            p for p, c in individual_estimations.items()
            if c != float('inf') and c >= max_estimated_count * self.CANDIDATE_THRESHOLD_RATIO
        })

        if not candidate_prices:
                  top_price = max(individual_estimations, key=individual_estimations.get)
                  candidate_prices = {top_price}

        # --- Stage 3: Pruned Exact Verification (Crossover + Mutation) ---
        final_counts = {}
        # Sort candidates by estimate (desc) to maximize pruning effectiveness
        sorted_candidates = sorted(
            list(candidate_prices),
            key=lambda p: individual_estimations.get(p, 0),
            reverse=True
        )

        max_known_exact_count = 0
        for price in sorted_candidates:
            # --- 突变部分：基于已知计数的剪枝验证 ---
            # If the current candidate's estimate is already lower than a known exact count,
            # it cannot be the mode. We can safely prune it.
            if individual_estimations.get(price, 0) < max_known_exact_count:
                continue

            exact_count = self._get_exact_count_for_price(tags, price)
            if exact_count > 0:
                final_counts[price] = exact_count
                if exact_count > max_known_exact_count:
                    max_known_exact_count = exact_count
        
        # --- Final Result Analysis ---
        modes = []
        max_count = 0
        if final_counts:
            max_count = max(final_counts.values())
            modes = sorted([p for p, c in final_counts.items() if c == max_count])

        total_time_ms = self._calculate_total_time()
        
        verification_data = {
            "batch_estimations": batch_estimations,
            "candidate_batch_index": candidate_batch_index,
            "individual_estimations": individual_estimations,
            "sorted_candidates_for_verification": sorted_candidates,
            "final_exact_counts": final_counts,
        }

        return modes, max_count, total_time_ms, verification_data

    def _estimate_batch_count(self, tags: List[Tag], batch: List[int]) -> float:
        """Estimates the total number of tags within a price batch."""
        self.total_frames += 1
        frame_size = self.BATCH_ESTIMATION_FRAME_SIZE
        seed = generate_random_seed()
        min_p, max_p = batch[0], batch[-1]
        self.total_bits_sent_by_reader += min_p.bit_length() + max_p.bit_length() + frame_size.bit_length() + seed.bit_length()

        # --- [Physical Layer Simulation] ---
        slot_states = [0] * frame_size
        for t in tags:
            if min_p <= t.price <= max_p:
                slot_id = simple_hash(t.id, seed) % frame_size
                slot_states[slot_id] += 1
        # --- [End Simulation] ---

        self.total_tag_bits_sent += frame_size * 1

        num_empty_slots = slot_states.count(0)
        
        if num_empty_slots == 0:
            return float('inf') # High contention
        if num_empty_slots == frame_size:
            return 0.0

        # Cardinality estimation formula
        estimated_count = -frame_size * math.log(num_empty_slots / frame_size)
        return estimated_count

    def _estimate_individual_counts(self, tags: List[Tag], prices: List[int]) -> Dict[int, float]:
        """(Derived from Parent B) Estimates counts for individual prices."""
        self.total_frames += 1
        frame_size = self.INDIVIDUAL_ESTIMATION_FRAME_SIZE
        seed = generate_random_seed()
        
        price_to_bit_pos = {price: i for i, price in enumerate(prices)}
        num_prices_in_batch = len(prices)
        self.total_bits_sent_by_reader += seed.bit_length() + frame_size.bit_length() + num_prices_in_batch.bit_length()

        # --- [Physical Layer Simulation] ---
        # Each slot can hold a bitmap of responses for all prices in the batch
        bits_per_slot = num_prices_in_batch
        physical_frame = Frame(num_slots=frame_size, bits_per_slot=bits_per_slot)
        price_set = set(prices)
        for t in tags:
            if t.price in price_set:
                slot_id = simple_hash(t.id, seed) % frame_size
                bit_pos = price_to_bit_pos[t.price]
                reply_vector = 1 << bit_pos
                physical_frame.aggregate(slot_id, reply_vector)
        # --- [End Simulation] ---
        
        self.total_tag_bits_sent += frame_size * bits_per_slot

        counts = {}
        for price, bit_pos in price_to_bit_pos.items():
            # Reader reconstructs the logical bit vector for this price
            logical_vector = [((slot_data >> bit_pos) & 1) for slot_data in physical_frame.slots]
            num_zeros = frame_size - sum(logical_vector)
            if num_zeros == 0:
                counts[price] = float('inf')
            elif num_zeros == frame_size:
                counts[price] = 0.0
            else:
                counts[price] = -frame_size * math.log(num_zeros / frame_size)
        return counts

    def _get_exact_count_for_price(self, tags: List[Tag], price: int) -> int:
        """(MODIFIED FOR COMPLIANCE) Gets the precise tag count for a single price."""
        identified_tags = set()
        frame_size = self.EXACT_COUNT_INITIAL_FRAME_SIZE

        # The loop must terminate based on observable events (no new tags, no collisions),
        # not on a priori knowledge of the total count.
        while True:
            self.total_frames += 1
            seed = generate_random_seed()
            self.total_bits_sent_by_reader += price.bit_length() + seed.bit_length() + frame_size.bit_length()

            # --- [Physical Layer Simulation] ---
            # This now correctly models tags responding based on the broadcast `price`.
            # The reader's logic does not have access to a pre-filtered set of tags.
            slots = [[] for _ in range(frame_size)]
            for t in tags:
                # Tag's logic: respond if its price matches the broadcast price
                # and it has not yet been identified (i.e., silenced).
                if t.price == price and t.id not in identified_tags:
                    slot_id = simple_hash(t.id, seed) % frame_size
                    slots[slot_id].append(t.id)
            # --- [End Simulation] ---

            self.total_tag_bits_sent += frame_size * 1

            newly_identified_this_frame = 0
            collision_detected = False
            # Reader's logic derives state from the physical simulation result.
            for slot in slots:
                if len(slot) == 1:  # 'Singleton' state
                    tag_id = slot[0]
                    # Check prevents re-counting (though simulation logic already handles this)
                    if tag_id not in identified_tags:
                        identified_tags.add(tag_id)
                        newly_identified_this_frame += 1
                elif len(slot) > 1:  # 'Collision' state
                    collision_detected = True

            # Simulate reader sending ACK/SILENCE commands to newly identified tags
            if newly_identified_this_frame > 0:
                self.total_bits_sent_by_reader += newly_identified_this_frame * 32  # Assuming 32-bit ID ACK

            # Compliant termination condition: If a frame has no new tags and no collisions,
            # the identification process for this price is complete.
            if newly_identified_this_frame == 0 and not collision_detected:
                break

            # Adaptive frame sizing based only on observable outcomes (collisions).
            if collision_detected:
                frame_size = min(frame_size * 2, 2048)
            # The non-compliant 'else' block for frame reduction is removed.
            # The break condition above now handles termination correctly.

        return len(identified_tags)

    def _calculate_total_time(self) -> float:
        """Calculates total execution time based on communication costs."""
        T_reader = self.total_bits_sent_by_reader * 0.025
        # Note: bits_per_slot varies. This is an aggregate calculation.
        T_tag = self.total_tag_bits_sent * 0.4
        T_overhead = (self.total_frames - 1) * 30 if self.total_frames > 1 else 0
        return T_reader + T_tag + T_overhead
