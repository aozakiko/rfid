import math
import random
from typing import List, Tuple, Dict
from collections import defaultdict
from base import Tag, Frame, generate_random_seed, IProtocol
from hash import simple_hash

class g1_1(IProtocol):
    """
    子代协议: 基于自适应优化的分批启发式众数查询 (Batched Heuristic with Adaptive Refinement)

    本协议融合了父代A(ModeQuerySEM)的分批并行思想和父代B(ModeQueryTCS)的非空时隙启发式指标，
    并通过引入全新的自适应帧长调整机制进行突变优化，旨在高效且准确地找出单一众数。

    协议流程:
    1.  粗筛阶段 (Coarse Screening): 将价格空间划分为多个批次，使用中等大小的帧对每个批次进行
        并行探测，并用非空时隙数作为初步评分，快速定位可能包含众数的高密度价格区间。
    2.  自适应精炼阶段 (Adaptive Refinement): 识别出在粗筛阶段中出现严重冲突（即非空时隙数
        接近帧长）的批次，并使用一个更大的帧长对这些批次进行重新探测，以获得更精确的评分，
        从而选出唯一的候选批次。
    3.  精确查找阶段 (Fine-grained Identification): 在唯一的候选批次内，对每个价格进行
        逐一探测，找出得分最高的具体价格作为最终的众数。
    """

    def execute(self, tags: List[Tag], price_min: int, price_max: int) -> Tuple[List[int], int, float, Dict]:
        total_frames = 0
        total_bits_sent_by_reader = 0
        total_tag_bits_sent = 0

        # --- 协议超参数 ---
        initial_batch_size = 100
        coarse_frame_size = 256
        refined_frame_size = 1024 # 用于高冲突批次的更大帧
        fine_grained_frame_size = 128
        contention_threshold_ratio = 0.95 # 冲突阈值

        # --- 读写器初始化 ---
        all_prices = list(range(price_min, price_max + 1))
        price_batches = [all_prices[i:i + initial_batch_size] for i in range(0, len(all_prices), initial_batch_size)]
        
        batch_scores = {}
        candidate_batch_index = -1

        # --- 交叉部分：融合父代A的分批思想与父代B的非空时隙启发式思想 ---
        # 阶段一：粗筛，快速定位高密度价格区间
        for i, batch in enumerate(price_batches):
            if not batch: continue
            
            # --- 读写器行为: 广播对价格批次 batch 的探测指令 ---
            total_frames += 1
            seed = generate_random_seed()
            # 广播指令包含: 批次范围(min, max), 帧长, 种子
            min_p, max_p = batch[0], batch[-1]
            total_bits_sent_by_reader += min_p.bit_length() + max_p.bit_length() + coarse_frame_size.bit_length() + seed.bit_length()

            # --- [物理层模拟] ---
            slot_states = [0] * coarse_frame_size
            for t in tags:
                if min_p <= t.price <= max_p:
                    slot_id = simple_hash(t.id, seed) % coarse_frame_size
                    slot_states[slot_id] += 1
            # --- [物理层模拟结束] ---

            # 探测阶段每个时隙回复1bit
            total_tag_bits_sent += coarse_frame_size * 1
            
            # --- 读写器分析: 使用父代B的启发式指标 ---
            num_non_empty_slots = sum(1 for count in slot_states if count > 0)
            batch_scores[i] = num_non_empty_slots

        # --- 突变部分：引入高冲突批次的自适应帧长重估全新机制 ---
        # 阶段二：对高冲突批次进行精炼
        
        # 识别高冲突批次
        high_contention_batches_indices = []
        contention_threshold = coarse_frame_size * contention_threshold_ratio
        for i, score in batch_scores.items():
            if score >= contention_threshold:
                high_contention_batches_indices.append(i)

        # 对高冲突批次使用更大帧长重新估算
        for i in high_contention_batches_indices:
            batch = price_batches[i]
            # --- 读写器行为: 广播对高冲突批次的精炼探测指令 ---
            total_frames += 1
            seed = generate_random_seed()
            min_p, max_p = batch[0], batch[-1]
            total_bits_sent_by_reader += min_p.bit_length() + max_p.bit_length() + refined_frame_size.bit_length() + seed.bit_length()
            
            # --- [物理层模拟] ---
            slot_states = [0] * refined_frame_size
            for t in tags:
                if min_p <= t.price <= max_p:
                    slot_id = simple_hash(t.id, seed) % refined_frame_size
                    slot_states[slot_id] += 1
            # --- [物理层模拟结束] ---

            # 探测阶段每个时隙回复1bit
            total_tag_bits_sent += refined_frame_size * 1
            
            # 更新评分为更精确的值
            refined_score = sum(1 for count in slot_states if count > 0)
            batch_scores[i] = refined_score

        # 选出最终的候选批次
        if batch_scores:
            # 通过对分数进行降序排序来找到得分最高的批次
            sorted_batches = sorted(batch_scores.items(), key=lambda item: item[1], reverse=True)
            if sorted_batches:
                candidate_batch_index = sorted_batches[0][0]

        # --- 交叉思想的延续：在筛选出的批次内进行精确查找 ---
        # 阶段三：在候选批次内进行精确查找
        price_to_score_map = {}
        if candidate_batch_index != -1:
            candidate_batch = price_batches[candidate_batch_index]
            for p in candidate_batch:
                total_frames += 1
                seed = generate_random_seed()
                total_bits_sent_by_reader += p.bit_length() + fine_grained_frame_size.bit_length() + seed.bit_length()

                # --- [物理层模拟] ---
                slot_states = [0] * fine_grained_frame_size
                for t in tags:
                    if t.price == p:
                        slot_id = simple_hash(t.id, seed) % fine_grained_frame_size
                        slot_states[slot_id] += 1
                # --- [物理层模拟结束] ---
                
                # 探测阶段每个时隙回复1bit
                total_tag_bits_sent += fine_grained_frame_size * 1
                
                num_non_empty_slots = sum(1 for count in slot_states if count > 0)
                if num_non_empty_slots > 0:
                    price_to_score_map[p] = num_non_empty_slots

        # --- 结果汇总与时间计算 ---
        T_reader = total_bits_sent_by_reader * 0.025
        T_tag = total_tag_bits_sent * 0.4 
        T_overhead = (total_frames - 1) * 30 if total_frames > 1 else 0
        total_time_ms = T_reader + T_tag + T_overhead

        verification_data = {
            "batch_scores": batch_scores,
            "candidate_batch_index": candidate_batch_index,
            "final_scores": price_to_score_map
        }

        if not price_to_score_map:
            return [], 0, total_time_ms, verification_data

        max_score = max(price_to_score_map.values())
        modes = sorted([
            price for price, score in price_to_score_map.items() 
            if score == max_score
        ])
        
        return modes, max_score, total_time_ms, verification_data

import math
from typing import List, Tuple, Dict
from collections import defaultdict
from base import Tag, IProtocol, generate_random_seed
from hash import simple_hash

class g2_3(IProtocol):
    """
    子代协议：带冲突率感知的渐进式深化协议 (Progressive Deepening with Sizing-awareness Protocol)

    本协议通过创新的两阶段方法，高效且准确地解决大规模RFID环境下的众数查询问题。
    它深度融合了父代协议A和B的核心思想，并引入了全新的动态优化机制。
    """

    def execute(self, tags: List[Tag], price_min: int, price_max: int) -> Tuple[List[int], int, float, Dict]:
        
        # --- 协议超参数 ---
        # 阶段一：初筛时的基础帧大小
        BASE_FRAME_SIZE = 128
        # 阶段二：对候选者进行深化验证的轮次
        VERIFICATION_ROUNDS = 3
        # 阶段二：帧长调整时，冲突率的加权因子
        COLLISION_SENSITIVITY_FACTOR = 2.5
        # 候选者选择：选取分数最高的 top N 个价格进入下一轮
        TOP_K_CANDIDATES = 5

        # --- 时间成本计算器初始化 ---
        total_bits_sent_by_reader = 0
        total_slots = 0
        total_frames = 0
        
        # ==================================================================
        # --- 交叉部分：融合父代A的“逐价扫描”思想与父代B的“估算-验证”两阶段框架 ---
        # 阶段一: 快速启发式扫描 (Rapid Heuristic Scanning)
        # ==================================================================
        # 此阶段类似父代A，对每个价格进行一次探测。
        # 同时，它作为父代B两阶段框架的第一步，用于快速筛选，而不是直接得出结论。
        # 我们不仅记录“非空时隙数”（父代A的核心指标），还额外记录“冲突时隙数”，为“突变”部分提供决策依据。
        
        price_to_metrics = {} # 存储 {price: {'score': score, 'collisions': collision_count}}

        for p in range(price_min, price_max + 1):
            total_frames += 1
            seed = generate_random_seed()
            
            # 读写器广播 Select(p), frame_size, seed
            total_bits_sent_by_reader += p.bit_length() + BASE_FRAME_SIZE.bit_length() + seed.bit_length()

            # --- [物理层模拟] ---
            slot_states = [0] * BASE_FRAME_SIZE # 0: 空闲, 1: 成功, >1: 冲突
            for t in tags:
                if t.price == p:
                    slot_id = simple_hash(t.id, seed) % BASE_FRAME_SIZE
                    slot_states[slot_id] += 1
            # --- [物理层模拟结束] ---
            
            total_slots += BASE_FRAME_SIZE

            # --- 读写器分析聚合信号 ---
            num_non_empty_slots = 0
            num_collision_slots = 0
            for count in slot_states:
                if count > 0:
                    num_non_empty_slots += 1
                if count > 1:
                    num_collision_slots += 1
            
            if num_non_empty_slots > 0:
                price_to_metrics[p] = {
                    "score": num_non_empty_slots, 
                    "collisions": num_collision_slots
                }

        # --- 候选者筛选 ---
        if not price_to_metrics:
            # 如果没有探测到任何标签，提前终止
            T_reader = total_bits_sent_by_reader * 0.025
            T_tag = total_slots * 1 * 0.4 # 探测阶段每个时隙回复1bit
            T_overhead = (total_frames - 1) * 30 if total_frames > 1 else 0
            total_time_ms = T_reader + T_tag + T_overhead
            return [], 0, total_time_ms, {"phase1_metrics": {}, "phase2_refined_scores": {}}

        # 按初筛分数（非空时隙数）降序排序，选出Top-K候选者
        sorted_candidates = sorted(
            price_to_metrics.items(), 
            key=lambda item: item[1]['score'], 
            reverse=True
        )
        candidate_prices = [item[0] for item in sorted_candidates[:TOP_K_CANDIDATES]]

        # ==================================================================
        # --- 突变部分：引入“基于冲突率的自适应帧长调整”和“渐进式深化”验证机制 ---
        # 阶段二: 渐进式深化验证 (Progressive Deepening Verification)
        # ==================================================================
        # 此机制在父代A和B中均未出现。
        # 父代A无验证阶段。父代B的验证阶段（_get_exact_count）虽然也自适应调整帧长，
        # 但它是在多轮识别循环中“被动地”响应碰撞，且目标是100%精确计数。
        # 本协议的创新点在于，在验证阶段开始前，就利用第一阶段收集的“冲突率”信息，
        # “主动地”、“前瞻性地”为每个候选者计算一个量身定制的、更优的初始帧长。
        # 随后，通过多轮（但非穷尽）的验证来获取一个更稳定、更可靠的启发式分数，
        # 而非追求绝对精确，从而在开销和准确性之间取得更好的平衡。
        
        refined_scores = defaultdict(list)

        for p_candidate in candidate_prices:
            metrics = price_to_metrics[p_candidate]
            score = metrics['score']
            collisions = metrics['collisions']

            # --- 核心突变逻辑 ---
            # 根据初筛阶段的冲突率计算动态帧长。冲突率越高，说明标签越多，需要的帧长越大。
            collision_rate = collisions / score if score > 0 else 0
            dynamic_frame_size = math.ceil(
                BASE_FRAME_SIZE * (1 + COLLISION_SENSITIVITY_FACTOR * collision_rate)
            )
            # 确保帧长在一个合理的范围内
            dynamic_frame_size = max(32, min(dynamic_frame_size, 4096))
            
            # --- 渐进式深化 ---
            # 使用动态计算出的帧长，进行多轮验证以平滑随机哈希带来的影响
            for _ in range(VERIFICATION_ROUNDS):
                total_frames += 1
                seed = generate_random_seed()
                
                total_bits_sent_by_reader += p_candidate.bit_length() + dynamic_frame_size.bit_length() + seed.bit_length()

                # --- [物理层模拟] ---
                slot_states = [0] * dynamic_frame_size
                for t in tags:
                    if t.price == p_candidate:
                        slot_id = simple_hash(t.id, seed) % dynamic_frame_size
                        slot_states[slot_id] += 1
                # --- [物理层模拟结束] ---
                
                total_slots += dynamic_frame_size
                
                num_non_empty_slots = sum(1 for count in slot_states if count > 0)
                refined_scores[p_candidate].append(num_non_empty_slots)
        
        # --- 最终处理 (Post-processing) ---
        final_scores = {
            price: sum(scores) / len(scores) 
            for price, scores in refined_scores.items()
        }

        # --- 时间成本计算 ---
        T_reader = total_bits_sent_by_reader * 0.025
        # 假设两个阶段每个时隙都只回复1bit用于探测
        T_tag = total_slots * 1 * 0.4 
        T_overhead = (total_frames - 1) * 30 if total_frames > 1 else 0
        total_time_ms = T_reader + T_tag + T_overhead
        
        # --- 找出得分最高的众数 ---
        if not final_scores:
            # 如果候选者验证后分数都为0，则退回使用第一阶段的结果
            final_scores = {p: m['score'] for p, m in price_to_metrics.items()}

        if not final_scores:
             return [], 0, total_time_ms, {"phase1_metrics": price_to_metrics, "phase2_refined_scores": final_scores}

        max_score = max(final_scores.values())
        modes = sorted([
            price for price, score in final_scores.items() 
            if score == max_score
        ])

        verification_data = {
            "phase1_metrics": price_to_metrics,
            "phase2_candidates": candidate_prices,
            "phase2_refined_scores": final_scores
        }

        return modes, max_score, total_time_ms, verification_data
