提示词：任务: 作为一名资深的RFID协议设计与优化专家，您的任务是基于给定的两个父代RFID协议（不要引入任何其他协议），创造一个全新的、功能完整的子代协议。该协议的核心目标是解决一个更具挑战性的 多众数查询问题。在此场景下，100000个标签（其价格属性值范围为1-1000）中存在多个频率相近的“热门”价格（众数），您的协议需要至少准确找出一个。

核心设计准则 (必须严格遵守):
1、强制交叉 (Mandatory Crossover): 新协议的设计必须体现出对两个父代协议核心思想的深度融合。您必须从父代A和父代B的设计哲学中汲取灵感，并有机地结合它们的优势，形成一个逻辑自洽的全新策略。
2、强制突变 (Mandatory Mutation): 在融合父代思想的基础上，新协议必须引入至少一项在两个父代中都未曾出现过的、全新的优化机制或逻辑。
3、准确率优先 (Accuracy First): 协议的逻辑正确性是最高优先级。最终方案必须是健壮的，能够稳定且准确地找出价格属性的众数。
4、信息隔离原则 (Information Isolation Principle):
    读写器 (Reader): 读写器默认已知环境中所有标签的ID列表作为全局背景信息。在协议执行期间，读写器可以向标签广播任何参数。其所有决策和分析，必须严格且仅依据以下信息：1) 它自己广播的历史参数；2) 从天线接收到的匿名聚合信号（即frame帧，一系列比特流）；3) 已知的全局标签ID列表。严禁在代码逻辑中直接或间接访问任何单个标签的私有属性（如 tag.price）。
    标签 (Tag): 标签的响应行为严格且仅依据以下信息：1) 自身的私有数据（如ID和价格）；2) 当前从读写器接收到的广播指令。严禁标签获取其他任何标签的信息或全局通信结果。
5、执行时间计算：你需要在代码中计算你设计协议的执行时间，其计算公式为:总执行时间公式: TotalTime = T_reader + T_tag + T_overhead
T_reader (ms) = total_bits_sent_by_reader × 0.025
T_tag (ms) = total_slots × bits_per_slot × 0.4
T_overhead (ms) = (total_frames - 1) × 30

# protocol/ModeQueryEE.py

import math
import time
from typing import List, Tuple, Dict
from base import Tag, IProtocol

class ModeQueryEE(IProtocol):
    """
    修正版协议：基于精确计数的“逐价清点”众数查询法
    
    本协议的核心思想保持不变：对每个可能的价格，通过反复寻找并计数拥有该价格
    的标签，来获得该价格的精确数量。此版本经过重构，严格遵守了信息隔离原则
    和逻辑正确性标准。
    
    优点:
    - 正确性极高：由于是精确计数，最终结果100%正确。
    
    缺点:
    - 效率极低：执行时间与 (价格范围 * 众数平均数量) 成正比。
    """

    def execute(self, tags: List[Tag], price_min: int, price_max: int) -> Tuple[List[int], int, float, Dict]:
        total_time_ms = 0.0
        price_exact_counts = {}

        # --- 协议超参数 (用于EE子协议模拟) ---
        EE_FRAME_SLOTS = 16
        TAG_ID_BITS = 32
        TIME_PER_EE_FRAME_MS = 58.8  # (2*0.2*32*1) + (16*1.0) + 30.0 = 58.8 ms
        TIME_PER_SILENT_CMD_MS = 6.4

        # --- 读写器逻辑: 遍历所有价格并发起计数指令 ---
        for p in range(price_min, price_max + 1):
            
            # --- [物理层模拟] ---
            # 读写器广播一个高级指令：“开始对价格p进行精确计数”。
            # 以下代码块模拟了物理层执行该指令的全过程，读写器无法看到内部细节。
            
            # 1. 物理层根据读写器广播的p，确定响应的标签群体
            current_price_tags_dict = {t.id: t for t in tags if t.price == p}
            
            count_for_p = 0
            time_cost_for_p = 0.0
            
            if current_price_tags_dict:
                # 2. 物理层反复执行EE子协议来寻找并“静默”最小ID的标签
                while current_price_tags_dict:
                    # a. 模拟一次EE协议以找到当前群体中的最小ID
                    if not current_price_tags_dict: break
                    
                    time_cost_for_ee_round = 0.0
                    all_ids = list(current_price_tags_dict.keys())
                    l_bound, h_bound = min(all_ids), max(all_ids) + 1

                    while (h_bound - l_bound) > 1:
                        time_cost_for_ee_round += TIME_PER_EE_FRAME_MS
                        d = (h_bound - l_bound) / EE_FRAME_SLOTS
                        if d == 0: break

                        min_slot_idx_found = float('inf')
                        for tag_id in current_price_tags_dict:
                            if l_bound <= tag_id < h_bound:
                                slot_idx = math.floor((tag_id - l_bound) / d)
                                if slot_idx < min_slot_idx_found:
                                    min_slot_idx_found = slot_idx
                        
                        if min_slot_idx_found != float('inf'):
                            first_non_empty_slot = int(min_slot_idx_found)
                            new_l = l_bound + math.ceil(d * first_non_empty_slot)
                            new_h = l_bound + math.ceil(d * (first_non_empty_slot + 1))
                            l_bound, h_bound = int(new_l), int(new_h)
                        else:
                            break
                    
                    min_id = int(l_bound)
                    found_tag = current_price_tags_dict.get(min_id)
                    time_cost_for_p += time_cost_for_ee_round

                    # b. 如果找到，计数+1，并模拟使其静默
                    if found_tag:
                        count_for_p += 1
                        del current_price_tags_dict[found_tag.id]
                        time_cost_for_p += TIME_PER_SILENT_CMD_MS
                    else:
                        # 理论上应该总能找到，除非l_bound计算有误或字典为空
                        break
            
            # --- [物理层模拟结束] ---

            # --- 读写器接收聚合结果并更新状态 ---
            # 读写器只得到一个最终计数值和总开销
            if count_for_p > 0:
                price_exact_counts[p] = count_for_p
                total_time_ms += time_cost_for_p
        
        # --- 读写器在本地进行最终结果分析 ---
        max_count = 0
        modes = []
        
        if price_exact_counts:
            max_count = max(price_exact_counts.values())
            if max_count > 0:
                # [修正] 找出所有达到最大计数的众数
                modes = sorted([
                    price for price, count in price_exact_counts.items() 
                    if count == max_count
                ])

        verification_data = {
            "ee_frame_slots": EE_FRAME_SLOTS,
            "final_exact_counts": price_exact_counts
        }
        
        return modes, max_count, total_time_ms, verification_data

import math
from typing import List, Tuple, Dict
from collections import defaultdict
from base import Tag, Frame, generate_random_seed, IProtocol,simple_hash



class ModeQueryTKQ(IProtocol):
    """
    实现了基于TKQ思想的价格众数查询协议。
    """
    
    def execute(self, tags: List[Tag], price_min: int, price_max: int) -> Tuple[List[int], int, float, Dict]:
        # *** 修改点 1：更新了返回类型以包含用于验证的字典 (Dict) ***
        total_time_ms = 0.0
        
        # ----------- 阶段一: 读写器初始化 -----------
        num_price_values = price_max - price_min + 1
        frame_size = num_price_values  # 帧长f设置为价格值的总数
        sog_length_v = 32              # SOG字符串长度
        omega_w = 1.2897               # 估算器常量
        num_rounds = 3                 # 为提高估算稳定性，执行多轮
        
        price_stats = {p: {"sum_L": 0.0, "num_slots": 0, "estimated_n": 0.0} 
                       for p in range(price_min, price_max + 1)}

        # --- 多轮估算循环 ---
        for _ in range(num_rounds):
            # ----------- 阶段二: 指令广播 -----------
            seed_r = generate_random_seed()
            
            bits_to_send = (seed_r.bit_length() + frame_size.bit_length() + sog_length_v.bit_length())
            time_send_ms = bits_to_send * 0.025
            total_time_ms += time_send_ms
            total_time_ms += 30.0 # 帧间开销

            # ----------- 阶段三: 标签处理 (模拟) -----------
            tag_replies = []
            for t in tags:
                # 1. 生成SOG字符串
                random_bits = simple_hash(t.id, seed_r)
                
                # --- [内联 _find_leading_zeros 逻辑] ---
                mu = 0
                count = 0
                for i in range(sog_length_v - 1, -1, -1):
                    if (random_bits >> i) & 1:
                        mu = count
                        break
                    count += 1
                else:
                    mu = sog_length_v
                # --- [内联结束] ---

                sog_string = (1 << (sog_length_v - 1 - mu)) if mu < sog_length_v else 0

                # 2. 根据价格选择时隙
                slot_id = simple_hash(t.price, seed_r) % frame_size
                tag_replies.append({'slot_id': slot_id, 'sog_string': sog_string})

            # ----------- 阶段四: 物理帧聚合与回复时间 -----------
            physical_frame = Frame(num_slots=frame_size, bits_per_slot=sog_length_v)
            for reply in tag_replies:
                physical_frame.aggregate(reply['slot_id'], reply['sog_string'])
                
            time_reply_ms = frame_size * sog_length_v * 0.4
            total_time_ms += time_reply_ms

            # ----------- 阶段五: 结果分析 -----------
            # 1. 预测价格到时隙的映射，识别同构时隙
            slot_to_prices_map = defaultdict(list)
            for p in range(price_min, price_max + 1):
                slot_idx = simple_hash(p, seed_r) % frame_size
                slot_to_prices_map[slot_idx].append(p)

            # 2. 处理同构时隙
            for slot_idx, prices in slot_to_prices_map.items():
                if len(prices) == 1:
                    price_val = prices[0]
                    combined_signal = physical_frame.slots[slot_idx]
                    
                    # b. 计算前导'1'的长度 L
                    # --- [内联 _find_leading_ones 逻辑] ---
                    L = 0
                    count = 0
                    for i in range(sog_length_v - 1, -1, -1):
                        if not ((combined_signal >> i) & 1):
                            L = count
                            break
                        count += 1
                    else:
                        L = sog_length_v
                    # --- [内联结束] ---
                    
                    # c. 更新统计数据
                    price_stats[price_val]["sum_L"] += L
                    price_stats[price_val]["num_slots"] += 1
                    
                    # d. 重新估算数量
                    s_p = price_stats[price_val]["num_slots"]
                    if s_p > 0:
                        avg_L = price_stats[price_val]["sum_L"] / s_p
                        estimated_n = omega_w * (2 ** avg_L)
                        price_stats[price_val]["estimated_n"] = estimated_n

        # ----------- 最终处理 (Post-processing) -----------
        # *** 修改点 2：创建用于验证的字典 ***
        verification_data = {
            "frame_size": frame_size,
            "bits_per_slot": sog_length_v
        }

        if not price_stats:
            return [], 0, total_time_ms, verification_data

        max_count = 0
        for stats in price_stats.values():
            if stats["estimated_n"] > max_count:
                max_count = stats["estimated_n"]
        
        mode_prices = []
        if max_count > 0:
            all_modes = sorted([price for price, stats in price_stats.items() 
                                if math.isclose(stats["estimated_n"], max_count)])
            if all_modes:
                # 只返回最大的众数
                mode_prices = [all_modes[-1]]
        
        # *** 修改点 3：在返回元组中增加验证字典 ***
        return mode_prices, math.ceil(max_count), total_time_ms, verification_data
