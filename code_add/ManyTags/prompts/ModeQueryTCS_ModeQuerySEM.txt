核心设计准则 (必须严格遵守):
1、强制交叉 (Mandatory Crossover): 新协议的设计必须体现出对两个父代协议核心思想的深度融合。您必须从父代A和父代B的设计哲学中汲取灵感，并有机地结合它们的优势，形成一个逻辑自洽的全新策略。

3、准确率优先 (Accuracy First): 协议的逻辑正确性是最高优先级。最终方案必须是健壮的，能够稳定且准确地找出价格属性的众数。
4、信息隔离原则 (Information Isolation Principle):
    读写器 (Reader): 读写器默认已知环境中所有标签的ID列表作为全局背景信息。在协议执行期间，读写器可以向标签广播任何参数。其所有决策和分析，必须严格且仅依据以下信息：1) 它自己广播的历史参数；2) 从天线接收到的匿名聚合信号（即frame帧，一系列比特流）；3) 已知的全局标签ID列表。严禁在代码逻辑中直接或间接访问任何单个标签的私有属性（如 tag.price）。
    标签 (Tag): 标签的响应行为严格且仅依据以下信息：1) 自身的私有数据（如ID和价格）；2) 当前从读写器接收到的广播指令。严禁标签获取其他任何标签的信息或全局通信结果。
5、执行时间计算：你需要在代码中计算你设计协议的执行时间，其计算公式为:总执行时间公式: TotalTime = T_reader + T_tag + T_overhead

T_tag (ms) = total_slots × bits_per_slot × 0.4
T_overhead (ms) = (total_frames - 1) × 30

# protocol/ModeQueryTCS.py

from typing import List, Tuple, Dict
from collections import defaultdict
from base import Tag, IProtocol, generate_random_seed,simple_hash

class ModeQueryTCS(IProtocol):
    """
    修正版协议：基于鲁棒启发式指标的TCS众数查询
    
    本协议实现了TCS（Tag Collision Separation）的基本思想，但进行了关键修正以
    保证其逻辑正确性并严格遵守信息隔离原则。协议迭代所有可能的价格，并为每个
    价格运行一次独立的帧通信，通过一个更鲁棒的启发式指标——“非空时隙数”
    （成功数+冲突数），来评估每个价格下的标签群体规模，从而找出最可能的众数。
    """
    
    def execute(self, tags: List[Tag], price_min: int, price_max: int) -> Tuple[List[int], int, float, Dict]:
        total_frames = 0
        total_bits_sent_by_reader = 0
        total_tag_bits_sent = 0

        # --- 协议超参数 ---
        frame_size = 128
        
        # [修正] 使用更有意义的变量名 price_to_score_map
        price_to_score_map = {}

        # ----------- 迭代查询所有价格 -----------
        for p in range(price_min, price_max + 1):
            
            # --- 读写器行为: 广播对价格 p 的探测指令 ---
            total_frames += 1
            seed_s = generate_random_seed()
            # 读写器广播Select(p), frame_size, seed
            total_bits_sent_by_reader += p.bit_length() + frame_size.bit_length() + seed_s.bit_length()

            # --- [物理层模拟] ---
            # 读写器无法访问标签私有数据。此代码块模拟所有标签的匿名、聚合响应。
            # 读写器能观察到的最终结果是 slot_states 数组。
            slot_states = [0] * frame_size  # 0: 空闲, 1: 成功, >1: 冲突
            
            # 1. 标签根据广播的 p 和自身的 price 判断是否响应
            # 2. 响应的标签根据广播的 seed 和自身的 id 计算时隙并回应
            for t in tags:
                if t.price == p:
                    slot_id = simple_hash(t.id, seed_s) % frame_size
                    slot_states[slot_id] += 1
            # --- [物理层模拟结束] ---

            # 探测阶段，每个时隙回复1bit即可
            total_tag_bits_sent += frame_size * 1

            # --- 读写器分析聚合信号 ---
            # [修正] 采用更鲁棒的“非空时隙数”作为评分
            num_non_empty_slots = sum(1 for count in slot_states if count > 0)
            
            if num_non_empty_slots > 0:
                price_to_score_map[p] = num_non_empty_slots

        # --- 最终处理 (Post-processing) ---
        
        # --- 时间成本计算 ---
        T_reader = total_bits_sent_by_reader * 0.025
        T_tag = total_tag_bits_sent * 0.4
        T_overhead = (total_frames - 1) * 30 if total_frames > 1 else 0
        total_time_ms = T_reader + T_tag + T_overhead

        verification_data = {
            "frame_size": frame_size,
            "heuristic_metric": "non_empty_slots",
            "final_scores": price_to_score_map
        }
        
        if not price_to_score_map:
            return [], 0, total_time_ms, verification_data

        # --- 找出得分最高的众数 ---
        max_score = max(price_to_score_map.values())
        
        # [修正] 返回所有并列的众数，并确保排序
        modes = sorted([
            price for price, score in price_to_score_map.items() 
            if score == max_score
        ])
        
        # 返回众数列表和它们对应的最高分
        return modes, max_score, total_time_ms, verification_data

# protocol/ModeQuerySEM.py

import math
import random
from typing import List, Tuple, Dict
from collections import defaultdict
from base import Tag, Frame, generate_random_seed, IProtocol,simple_hash

class ModeQuerySEM(IProtocol):
    """
    修正版协议：分批次SEM众数估算协议 (Batched SEM Mode Estimation Protocol)

    本协议实现了分批次SEM（Slot-Encoded Multiplexing）的估算思想，并进行了
    关键修正以保证其逻辑正确性并严格遵守信息隔离原则。协议将所有价格分批处理，
    在每个批次内，并行地对多个价格的标签数量进行估算。最终，协议找出估算值
    最高的若干个价格作为最可能的众数。
    """
    
    def execute(self, tags: List[Tag], price_min: int, price_max: int) -> Tuple[List[int], int, float, Dict]:
        total_frames = 0
        total_bits_sent_by_reader = 0
        total_tag_bits_sent = 0

        # --- 协议超参数 ---
        batch_size = 100
        frame_size = 512 # 每批次使用的帧长固定
        
        # --- 读写器初始化 ---
        all_prices = list(range(price_min, price_max + 1))
        price_batches = [all_prices[i:i + batch_size] for i in range(0, len(all_prices), batch_size)]
        
        final_counts = {}

        # --- 逐批估算 ---
        for batch in price_batches:
            if not batch: continue
            
            # --- 读写器行为: 广播对一批价格的并行估算指令 ---
            total_frames += 1
            num_prices_in_batch = len(batch)
            seed = generate_random_seed()
            # 读写器知道自己要如何解析回复，因此在本地创建此映射
            price_to_bit_pos_map = {price: i for i, price in enumerate(batch)}
            
            # 广播 seed, frame_size, num_prices_in_batch (或整个batch的价格列表)
            total_bits_sent_by_reader += (seed.bit_length() + frame_size.bit_length() + num_prices_in_batch.bit_length())

            # --- [物理层模拟] ---
            # 读写器无法访问标签。此代码块模拟所有标签根据收到的指令
            # (seed, price_map) 产生匿名的、聚合的响应帧。
            physical_frame = Frame(num_slots=frame_size, bits_per_slot=num_prices_in_batch)
            batch_price_set = set(batch)
            for t in tags:
                if t.price in batch_price_set:
                    slot_id = simple_hash(t.id, seed) % frame_size
                    bit_pos = price_to_bit_pos_map[t.price]
                    reply_vector = 1 << bit_pos
                    physical_frame.aggregate(slot_id, reply_vector)
            # --- [物理层模拟结束] ---

            # 累加标签通信开销
            total_tag_bits_sent += frame_size * num_prices_in_batch
            
            # --- 读写器分析聚合的 Frame 对象 ---
            # 读写器只接收到 physical_frame，并使用它自己的映射表来解析
            for price, bit_pos in price_to_bit_pos_map.items():
                logical_bit_vector = [((slot_data >> bit_pos) & 1) for slot_data in physical_frame.slots]
                num_zeros = frame_size - sum(logical_bit_vector)
                
                if num_zeros == frame_size:
                    final_counts[price] = 0
                elif num_zeros == 0:
                    final_counts[price] = float('inf')
                else:
                    estimated_count = -frame_size * math.log(num_zeros / frame_size)
                    final_counts[price] = estimated_count

        # --- 结果汇总与时间计算 ---
        T_reader = total_bits_sent_by_reader * 0.025
        T_tag = total_tag_bits_sent * 0.4
        T_overhead = (total_frames - 1) * 30 if total_frames > 1 else 0
        total_time_ms = T_reader + T_tag + T_overhead
        
        verification_data = {
            "frame_size_per_batch": frame_size,
            "batch_size": batch_size,
            "final_estimated_counts": final_counts
        }
        
        if not final_counts:
            return [], 0, total_time_ms, verification_data

        # --- 找出估算值最高的众数 ---
        max_count = 0
        # 优先处理inf的情况
        if float('inf') in final_counts.values():
            max_count = float('inf')
        else:
            # 在所有有限的计数值中寻找最大值
            valid_counts = [c for c in final_counts.values()]
            if valid_counts:
                max_count = max(valid_counts)
        
        modes = []
        if max_count > 0:
            # [修正] 返回所有并列的众数
            if max_count == float('inf'):
                modes = sorted([price for price, count in final_counts.items() if count == float('inf')])
            else:
                # 使用 math.isclose 处理浮点数比较
                modes = sorted([price for price, count in final_counts.items() if math.isclose(count, max_count)])
        
        # 将估算值转换为整数以便返回，或保持inf
        final_max_count = math.ceil(max_count) if max_count != float('inf') else float('inf')
        
        return modes, final_max_count, total_time_ms, verification_data
