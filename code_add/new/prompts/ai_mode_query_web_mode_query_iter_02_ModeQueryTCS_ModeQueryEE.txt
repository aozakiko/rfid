核心设计准则（必须严格遵守）：
1. 强制交叉（Mandatory Crossover）：新协议的设计必须体现对父代 A 和父代 B 核心思想的深度融合，而不是简单复制其中一个父代。
2. 强制突变（Mandatory Mutation）：在融合父代思想的基础上，新协议必须引入至少一项父代中没有的优化机制，例如动态帧长、分层采样、草图估计、置信区间修正、候选桶重检、早停策略或多阶段校验。

5. 信息隔离原则（Information Isolation Principle）：
   - 读写器（Reader）：读写器默认已知环境中所有标签 ID 列表作为全局背景信息。协议执行期间，读写器可以广播参数，但它的所有决策和分析只能依据：1) 自己广播过的历史参数；2) 从天线接收到的匿名聚合信号（frame/bit stream/collision sketch）；3) 已知标签 ID 列表。严禁在读写器决策逻辑中直接或间接访问单个标签私有属性，例如 tag.price。
   - 标签（Tag）：标签响应只能依据：1) 自身私有数据（ID 和数值属性）；2) 当前读写器广播指令。标签不得读取其他标签信息或全局通信结果。
6. 执行时间计算：代码中必须计算协议执行时间，推荐公式为：
   TotalTime = T_reader + T_tag + T_overhead
   T_reader (ms) = total_bits_sent_by_reader * 0.025
   T_tag (ms) = total_slots * bits_per_slot * 0.4

# protocol/ModeQueryTCS.py

from typing import List, Tuple, Dict
from collections import defaultdict
from base import Tag, IProtocol, generate_random_seed,simple_hash

class ModeQueryTCS(IProtocol):
    """
    修正版协议：基于鲁棒启发式指标的TCS众数查询
    
    本协议实现了TCS（Tag Collision Separation）的基本思想，但进行了关键修正以
    保证其逻辑正确性并严格遵守信息隔离原则。协议迭代所有可能的价格，并为每个
    价格运行一次独立的帧通信，通过一个更鲁棒的启发式指标——“非空时隙数”
    （成功数+冲突数），来评估每个价格下的标签群体规模，从而找出最可能的众数。
    """
    
    def execute(self, tags: List[Tag], price_min: int, price_max: int) -> Tuple[List[int], int, float, Dict]:
        total_frames = 0
        total_bits_sent_by_reader = 0
        total_tag_bits_sent = 0

        # --- 协议超参数 ---
        frame_size = 128
        
        # [修正] 使用更有意义的变量名 price_to_score_map
        price_to_score_map = {}

        # ----------- 迭代查询所有价格 -----------
        for p in range(price_min, price_max + 1):
            
            # --- 读写器行为: 广播对价格 p 的探测指令 ---
            total_frames += 1
            seed_s = generate_random_seed()
            # 读写器广播Select(p), frame_size, seed
            total_bits_sent_by_reader += p.bit_length() + frame_size.bit_length() + seed_s.bit_length()

            # --- [物理层模拟] ---
            # 读写器无法访问标签私有数据。此代码块模拟所有标签的匿名、聚合响应。
            # 读写器能观察到的最终结果是 slot_states 数组。
            slot_states = [0] * frame_size  # 0: 空闲, 1: 成功, >1: 冲突
            
            # 1. 标签根据广播的 p 和自身的 price 判断是否响应
            # 2. 响应的标签根据广播的 seed 和自身的 id 计算时隙并回应
            for t in tags:
                if t.price == p:
                    slot_id = simple_hash(t.id, seed_s) % frame_size
                    slot_states[slot_id] += 1
            # --- [物理层模拟结束] ---

            # 探测阶段，每个时隙回复1bit即可
            total_tag_bits_sent += frame_size * 1

            # --- 读写器分析聚合信号 ---
            # [修正] 采用更鲁棒的“非空时隙数”作为评分
            num_non_empty_slots = sum(1 for count in slot_states if count > 0)
            
            if num_non_empty_slots > 0:
                price_to_score_map[p] = num_non_empty_slots

        # --- 最终处理 (Post-processing) ---
        
        # --- 时间成本计算 ---
        T_reader = total_bits_sent_by_reader * 0.025
        T_tag = total_tag_bits_sent * 0.4
        T_overhead = (total_frames - 1) * 30 if total_frames > 1 else 0
        total_time_ms = T_reader + T_tag + T_overhead

        verification_data = {
            "frame_size": frame_size,
            "heuristic_metric": "non_empty_slots",
            "final_scores": price_to_score_map
        }
        
        if not price_to_score_map:
            return [], 0, total_time_ms, verification_data

        # --- 找出得分最高的众数 ---
        max_score = max(price_to_score_map.values())
        
        # [修正] 返回所有并列的众数，并确保排序
        modes = sorted([
            price for price, score in price_to_score_map.items() 
            if score == max_score
        ])
        
        # 返回众数列表和它们对应的最高分
        return modes, max_score, total_time_ms, verification_data

# protocol/ModeQueryEE.py

import math
import time
from typing import List, Tuple, Dict
from base import Tag, IProtocol

class ModeQueryEE(IProtocol):
    """
    修正版协议：基于精确计数的“逐价清点”众数查询法
    
    本协议的核心思想保持不变：对每个可能的价格，通过反复寻找并计数拥有该价格
    的标签，来获得该价格的精确数量。此版本经过重构，严格遵守了信息隔离原则
    和逻辑正确性标准。
    
    优点:
    - 正确性极高：由于是精确计数，最终结果100%正确。
    
    缺点:
    - 效率极低：执行时间与 (价格范围 * 众数平均数量) 成正比。
    """

    def execute(self, tags: List[Tag], price_min: int, price_max: int) -> Tuple[List[int], int, float, Dict]:
        total_time_ms = 0.0
        price_exact_counts = {}

        # --- 协议超参数 (用于EE子协议模拟) ---
        EE_FRAME_SLOTS = 16
        TAG_ID_BITS = 32
        TIME_PER_EE_FRAME_MS = 58.8  # (2*0.2*32*1) + (16*1.0) + 30.0 = 58.8 ms
        TIME_PER_SILENT_CMD_MS = 6.4

        # --- 读写器逻辑: 遍历所有价格并发起计数指令 ---
        for p in range(price_min, price_max + 1):
            
            # --- [物理层模拟] ---
            # 读写器广播一个高级指令：“开始对价格p进行精确计数”。
            # 以下代码块模拟了物理层执行该指令的全过程，读写器无法看到内部细节。
            
            # 1. 物理层根据读写器广播的p，确定响应的标签群体
            current_price_tags_dict = {t.id: t for t in tags if t.price == p}
            
            count_for_p = 0
            time_cost_for_p = 0.0
            
            if current_price_tags_dict:
                # 2. 物理层反复执行EE子协议来寻找并“静默”最小ID的标签
                while current_price_tags_dict:
                    # a. 模拟一次EE协议以找到当前群体中的最小ID
                    if not current_price_tags_dict: break
                    
                    time_cost_for_ee_round = 0.0
                    all_ids = list(current_price_tags_dict.keys())
                    l_bound, h_bound = min(all_ids), max(all_ids) + 1

                    while (h_bound - l_bound) > 1:
                        time_cost_for_ee_round += TIME_PER_EE_FRAME_MS
                        d = (h_bound - l_bound) / EE_FRAME_SLOTS
                        if d == 0: break

                        min_slot_idx_found = float('inf')
                        for tag_id in current_price_tags_dict:
                            if l_bound <= tag_id < h_bound:
                                slot_idx = math.floor((tag_id - l_bound) / d)
                                if slot_idx < min_slot_idx_found:
                                    min_slot_idx_found = slot_idx
                        
                        if min_slot_idx_found != float('inf'):
                            first_non_empty_slot = int(min_slot_idx_found)
                            new_l = l_bound + math.ceil(d * first_non_empty_slot)
                            new_h = l_bound + math.ceil(d * (first_non_empty_slot + 1))
                            l_bound, h_bound = int(new_l), int(new_h)
                        else:
                            break
                    
                    min_id = int(l_bound)
                    found_tag = current_price_tags_dict.get(min_id)
                    time_cost_for_p += time_cost_for_ee_round

                    # b. 如果找到，计数+1，并模拟使其静默
                    if found_tag:
                        count_for_p += 1
                        del current_price_tags_dict[found_tag.id]
                        time_cost_for_p += TIME_PER_SILENT_CMD_MS
                    else:
                        # 理论上应该总能找到，除非l_bound计算有误或字典为空
                        break
            
            # --- [物理层模拟结束] ---

            # --- 读写器接收聚合结果并更新状态 ---
            # 读写器只得到一个最终计数值和总开销
            if count_for_p > 0:
                price_exact_counts[p] = count_for_p
                total_time_ms += time_cost_for_p
        
        # --- 读写器在本地进行最终结果分析 ---
        max_count = 0
        modes = []
        
        if price_exact_counts:
            max_count = max(price_exact_counts.values())
            if max_count > 0:
                # [修正] 找出所有达到最大计数的众数
                modes = sorted([
                    price for price, count in price_exact_counts.items() 
                    if count == max_count
                ])

        verification_data = {
            "ee_frame_slots": EE_FRAME_SLOTS,
            "final_exact_counts": price_exact_counts
        }
        
        return modes, max_count, total_time_ms, verification_data
